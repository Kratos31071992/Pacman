cmake_minimum_required(VERSION 3.16)
project(Pacman VERSION 1.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(pacman_core src/Game.cpp)
target_include_directories(pacman_core PUBLIC src)

add_executable(Pacman src/main.cpp src/Terminal.cpp)
target_link_libraries(Pacman PRIVATE pacman_core)

if(MSVC)
    target_compile_options(pacman_core PRIVATE /W4)
    target_compile_options(Pacman PRIVATE /W4)
else()
    target_compile_options(pacman_core PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(Pacman PRIVATE -Wall -Wextra -Wpedantic)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(pacman_tests tests/GameTests.cpp)
    target_link_libraries(pacman_tests PRIVATE pacman_core)
    if(MSVC)
        target_compile_options(pacman_tests PRIVATE /W4)
    else()
        target_compile_options(pacman_tests PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    add_test(NAME pacman_game_logic COMMAND pacman_tests)
endif()
